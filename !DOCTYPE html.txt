      <div><label>Pick up / Leaving With Individuals</label><input data-k="arrival" type="time" required></div>

<script>
  var MAX_LEGS = 23;
  var $ = function (s, r) { return (r || document).querySelector(s); };
  var form = $("#f"), legsEl = $("#legs"), peopleEl = $("#people");
  var rosterReq = 0;                         // guards against out-of-order responses when switching homes / shifts

  function to12h(v) {                       // "15:05" -> "3:05 PM"
    if (!v || v === "NS") return v;
    var p = v.split(":"), h = +p[0], ap = h >= 12 ? "PM" : "AM";
    return ((h % 12) || 12) + ":" + p[1] + " " + ap;
  }
  function toSheetDate(v) { var p = v.split("-"); return p[1] + "/" + p[2] + "/" + p[0]; }

  function addLeg() {
    if (legsEl.children.length >= MAX_LEGS) return;
    var leg = $("#legTpl").content.firstElementChild.cloneNode(true);
    legsEl.appendChild(leg); refresh();
  }

  // Loads the names from the "<home> AM" / "<home> PM" tab and shows them as checkboxes.
  function renderPeople() {
    var code = $("#home").value, shift = $("#shift").value, req = ++rosterReq;
    peopleEl.innerHTML = "";
    if (!code || !shift) {
      peopleEl.innerHTML = '<p class="empty">Select a home and AM or PM to see its individuals.</p>';
      return;
    }
    peopleEl.innerHTML = '<p class="empty">Loading individuals...</p>';
    google.script.run
      .withSuccessHandler(function (names) {
        if (req !== rosterReq) return;
        peopleEl.innerHTML = "";
        if (!names || !names.length) {
          peopleEl.innerHTML = '<p class="empty">No individuals are listed for this home and trip yet.</p>';
          return;
        }
        var wrap = document.createElement("div");
        wrap.className = "people";
        names.forEach(function (n) {
          var lab = document.createElement("label");
          lab.className = "person";
          var cb = document.createElement("input");
          cb.type = "checkbox"; cb.className = "personBox"; cb.value = n;
          lab.appendChild(cb);
          lab.appendChild(document.createTextNode(n));
          wrap.appendChild(lab);
        });
        peopleEl.appendChild(wrap);
      })
      .withFailureHandler(function (err) {
        if (req !== rosterReq) return;
        peopleEl.innerHTML = '<p class="empty">Could not load individuals: ' +
          (err && err.message ? err.message : err) + '</p>';
      })
      .getHomeMembers(code, shift);
  }
  function selectedPeople() {
    return Array.prototype.map.call(peopleEl.querySelectorAll(".personBox:checked"), function (c) { return c.value; });
  }

  // Submit stays disabled while the form is incomplete OR the driver name/license do not match our records
  var driverBlocked = false;
  function refresh() {
    var ok = form.checkValidity() && !driverBlocked;
    $("#submit").disabled = !ok;
    $("#status").textContent = driverBlocked
      ? "Driver name and license do not match our records. Check your entries or contact ITadmin@onediversity.org."
      : (ok ? "Ready to submit." : "Complete every field to enable submit.");
  }

  // Driver's license: strip spaces/dashes, make capitals, first character a letter then 8 numbers,
  // and show how many of 9 characters are typed
  function updateLicCount() {
    var lic = $("#license"), n = lic.value.length, c = $("#licCount");
    c.textContent = n === 9 ? "9 of 9 characters - looks good!" : n + " of 9 characters typed";
    c.className = "count" + (n === 9 ? " good" : "");
  }
  $("#license").addEventListener("input", function () {
    var raw = this.value.toUpperCase(), clean = "";
    for (var i = 0; i < raw.length && clean.length < 9; i++) {
      var ch = raw.charAt(i);
      if (clean.length === 0) { if (/[A-Z]/.test(ch)) clean += ch; }   // first character: letter only
      else if (/[0-9]/.test(ch)) clean += ch;                           // the rest: numbers only
    }
    if (clean !== this.value) this.value = clean;
    updateLicCount();
  });
  form.addEventListener("reset", function () { setTimeout(updateLicCount, 0); });

  $("#home").addEventListener("change", renderPeople);
  $("#shift").addEventListener("change", renderPeople);
  form.addEventListener("input", refresh);
  form.addEventListener("change", refresh);
  form.addEventListener("focusout", function (e) { e.target.classList.add("touched"); });

  form.addEventListener("submit", function (e) {
    e.preventDefault();
    if (!form.checkValidity()) { form.reportValidity(); return; }
    var legs = Array.prototype.map.call(legsEl.children, function (l) {
      var g = function (k) { return $('[data-k="' + k + '"]', l).value.trim(); };
      return {
        arrival: to12h(g("arrival")),
        dropoff: to12h(g("dropoff")),
        signature: g("signature")
      };
    });
    var payload = {
      home: $("#home").value, shift: $("#shift").value, serviceDate: toSheetDate($("#serviceDate").value),
      license: $("#license").value.trim(),
      driverName: $("#driverName").value.trim(), driverSignature: $("#driverSignature").value.trim(),
      individuals: selectedPeople(),
      legs: legs
    };
    var btn = $("#submit"), msg = $("#msg");
    btn.disabled = true; btn.textContent = "Submitting..."; msg.className = "";
    showLoading();
    google.script.run
      .withSuccessHandler(function (r) {
        hideLoading();
        // Clear the form behind the scenes, then cover the screen with the success message.
        form.reset(); legsEl.innerHTML = ""; addLeg(); renderPeople();
        Array.prototype.forEach.call(form.querySelectorAll(".touched"), function (el) { el.classList.remove("touched"); });
        $("#serviceDate").value = todayStr();
        driverBlocked = false;
        btn.textContent = "Submit trip log"; msg.className = ""; refresh();
        $("#done").classList.add("show");
        document.body.style.overflow = "hidden";
        $("#again").focus();
      })
      .withFailureHandler(function (err) {
        hideLoading();
        var m = (err && err.message) ? err.message : String(err);
        // If the server rejects the driver name/license, show the popup too
        if (/does not match our records/i.test(m)) { driverBlocked = true; showDriverModal(); }
        msg.className = "err";
        msg.textContent = "Not submitted: " + m + " Your entries are still here. Fix the problem and submit again.";
        btn.textContent = "Submit trip log"; refresh();
      })
      .submitTripLog(payload);
  });

  // Loading screen: covers the page and warns if staff try to close the tab mid-submit
  function showLoading() {
    $("#loading").classList.add("show");
    document.body.style.overflow = "hidden";
    window.onbeforeunload = function (e) { e.preventDefault(); e.returnValue = ""; return ""; };
  }
  function hideLoading() {
    $("#loading").classList.remove("show");
    document.body.style.overflow = "";
    window.onbeforeunload = null;
  }

  // "Submit another trip log" button on the success screen
  $("#again").addEventListener("click", function () {
    $("#done").classList.remove("show");
    document.body.style.overflow = "";
    window.scrollTo({ top: 0 });
  });

  function todayStr() {
    var d = new Date(), m = ("0" + (d.getMonth() + 1)).slice(-2), day = ("0" + d.getDate()).slice(-2);
    return d.getFullYear() + "-" + m + "-" + day;
  }

  // ---- Live driver check (name + license vs. the Drivers tab) ----
  var driverReq = 0, driverTimer = null;
  function showDriverModal() {
    $("#driverModal").classList.add("show");
    document.body.style.overflow = "hidden";
    $("#dmFix").focus();
  }
  function hideDriverModal() {
    $("#driverModal").classList.remove("show");
    document.body.style.overflow = "";
  }
  function checkDriverNow() {
    var name = $("#driverName").value.trim(), lic = $("#license").value;
    var req = ++driverReq;
    if (!name || lic.length !== 9) { driverBlocked = false; refresh(); return; }
    google.script.run
      .withSuccessHandler(function (r) {
        if (req !== driverReq) return;               // ignore out-of-date answers
        driverBlocked = (r && r.status === "mismatch");
        refresh();
        if (driverBlocked) showDriverModal();
      })
      .withFailureHandler(function () { if (req === driverReq) { driverBlocked = false; refresh(); } })
      .checkDriver(name, lic);
  }
  function queueDriverCheck() {
    clearTimeout(driverTimer);
    driverBlocked = false; refresh();                // clear the old warning while they retype
    driverTimer = setTimeout(checkDriverNow, 500);
